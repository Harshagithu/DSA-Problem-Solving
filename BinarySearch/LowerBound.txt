

Input :- [1,2,3,6,7,9,10,11] target = 8

Output :- 5

func(arr,n,target)
{
 int low=0,high=n-1;
 int ans=n;
 while(low<=high)
 {
   int mid=(low+high)/2;
   if(arr[mid]>=target)
   {
     ans=mid;
     low=mid+1;
   }
   else
   {
     high=mid-1;
   }
 }
 return ans;
}

Time Complexity :- O(log2n)
Space Complexity :- O(1)
