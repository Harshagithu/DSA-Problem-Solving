

Input :- arr = [1,2,2,3,8,8,8,9,10,11,11,13] , target = 8

Output :- [4,6] (8 first occurred at 4th index and last occured at 6th index)


Brute Force :-
==============

func(arr,n,target)
{
 int first=-1,last=-1;
 for(int i=0;i<n;i++)
 {
  if(arr[i]==target)
  {
    if(first!=-1)
    {
      first=i;
    }
    last=i;
  }
 }
 return new int[]{first,last};
}

Time Complexity :- O(n)
Space Complexity :- O(1)

Optimal Approach:-
==================

firstOccurence(arr,n,target)
{
  int low=0,high=n-1;
  int first=-1;
  while(low<=high)
  {
    int mid=(low+high)/2;
    if(arr[mid]==target)
    {
      first=mid;
      high=mid-1;
    }
    else if(arr[mid]<target)
    {
      low=mid+1;
    }
    else
    {
      high=mid-1;
    }
  }
  return first;
}

lastOccurence(arr,n,target)
{
  int low=0,high=n-1;
  int last=-1;
  while(low<=high)
  {
    int mid=(low+high)/2;
    if(arr[mid]==target)
    {
      last=mid;
      low=mid+1;
    }
    else if(arr[mid]<target)
    {
      low=mid+1;
    }
    else
    {
      high=mid-1;
    }
  }
  return last;
}

func(arr,n,target)
{
 int first=firstOccurence(arr,n,target);
 if(first==-1) return new int[]{-1,-1};
 return new int[]{first,lastOccurence(arr,n,target)};
}

Time Complexity :- O(2*log2n)
Space Complexity :- O(1)
