

Input :- arr = [1,2,2,3,8,8,8,9,10,11,11,13] , target = 8

Output :- 3 (8 occurs 3 times).


Brute Force :-
==============

func(arr,n,target)
{
 int count=0;
 for(int i=0;i<n;i++)
 {
  if(arr[i]==target)
  {
    count++;
  }
 }
 return count;
}

Time Complexity :- O(n)
Space Complexity :- O(1)

Optimal Approach:-
==================

firstOccurence(arr,n,target)
{
  int low=0,high=n-1;
  int first=-1;
  while(low<=high)
  {
    int mid=(low+high)/2;
    if(arr[mid]==target)
    {
      first=mid;
      high=mid-1;
    }
    else if(arr[mid]<target)
    {
      low=mid+1;
    }
    else
    {
      high=mid-1;
    }
  }
  return first;
}

lastOccurence(arr,n,target)
{
  int low=0,high=n-1;
  int last=-1;
  while(low<=high)
  {
    int mid=(low+high)/2;
    if(arr[mid]==target)
    {
      last=mid;
      low=mid+1;
    }
    else if(arr[mid]<target)
    {
      low=mid+1;
    }
    else
    {
      high=mid-1;
    }
  }
  return last;
}

func(arr,n,target)
{
 int first=firstOccurence(arr,n,target);
 if(first==0)return 0;
 return lastOccurence(arr,n,target)-first+1;
}

Time Complexity :- O(2*log2n)
Space Complexity :- O(1)
