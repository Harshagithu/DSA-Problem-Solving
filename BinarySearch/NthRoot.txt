
Input :- n=3 m=27

Output :- 3

Brute Force:-
=============

solve(long ans,int n,int m)
{
  for(int i=1;i<=n;i++)
  {
   ans=ans*ans;
   if(ans>m) return 2;
  }
  if(ans==m) return 1;
  return 0;
}

func(n,m)
{
  for(int i=1;i<=m;i++)
  {
   long val=solve((long)i,n,m);
   if(val==1)
   {
     return i;
   }
  }
  return -1;
}

Time Complexity :- O(m*log2n)
Space Complexity :- O(1)


Optimal Approach :-
===================

solve(long ans,int n,int m)
{
  for(int i=1;i<=n;i++)
  {
   ans=ans*ans;
   if(ans>m) return 2;
  }
  if(ans==m) return 1;
  return 0;
}

func(n,m)
{
  long low=1,high=m;
  while(low<=high)
  {
   long mid=low+(high-low)/2;
   long val=solve(mid,n,m);
   if(val==1)
   {
     return i;
   }
   else if(val==0)
   {
     low=mid+1;
   }
   else
   {
     high=mid-1;
   }
  }
  return -1;
}

Time Complexity :- O(log2m*log2n)
Space Complexity :- O(1)
