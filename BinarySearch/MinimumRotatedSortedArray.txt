

Input :- arr = [3,4,5,1,2]

Output :- 1

Brute Force :-
==============

func(arr,n)
{
 int mini=Integer.MAX_VALUE;
 for(int i=0;i<n;i++)
 {
  mini=Math.min(mini,arr[i]);
 }
 return mini;
}

Time Complexity :- O(n)
Space Complexity :- O(1)

Optimal Approach :-
===================

func(arr,n)
{
 int ans=Integer.MAX_VALUE;
 int low=0,high=n-1;
 while(low<=high)
 {
  int mid=(low+high)/2;
  if(arr[low]<=arr[high)
  {
    if(ans<arr[low])
    {
      ans=arr[low];
      break;
    }
  }
  if(arr[low]<=arr[mid])
  {
   if(ans<arr[low])
   {
     ans=arr[low];
   }
   low=mid+1;
  }
  else
  {
   if(ans<arr[mid])
   {
     ans=arr[mid];
   }
   high=mid-1;
  }
 }
 return ans;
}

Time Complexity :- O(log2n)
Space Complexity :- O(1)
