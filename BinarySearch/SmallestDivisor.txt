int smallestDivisor(int[] nums, int threshold) {
        int n=nums.length;
        int maxi=Integer.MIN_VALUE;
        for(int i=0;i<n;i++)
        {
            maxi=Math.max(maxi,nums[i]);
        }
        int low=0,high=maxi;
        while(low<=high)
        {
            int mid=(low+high)/2;
            if(sumDiv(nums,n,mid)<=threshold)
            {
                high=mid-1;
            }
            else
            {
                low=mid+1;
            }
        }
        return low;
    }
    public int sumDiv(int nums[],int n,int divisor)
    {
        int sum=0;
        for(int i=0;i<n;i++)
        {
            sum+=Math.ceil((double)nums[i]/(double)divisor);
        }
        return sum;
    }

Time Complexity :- O(log2maxi*n) + O(n)
Space Complexity :- O(1)
