
Input :- arr[] = [1, 2, 4, 8, 9], k = 3

Output :- 3

Explanation :- The first cow can be placed at arr[0], the second at arr[2], and the third at arr[3].
The minimum distance between any two cows is 3 (between arr[0] and arr[2]), which is the maximum possible among all valid arrangements.

Brute Force :-
==============

int aggressiveCows(int[] arr, int k) {
        Arrays.sort(arr);
        int n=arr.length;
        int diff=arr[n-1]-arr[0];
        for(int i=1;i<=diff;i++)
        {
            if(isPossibleCows(arr,n,i,k))
            {
                continue;
            }
            else
            {
                return i-1;
            }
        }
        return diff;
    }
    public boolean isPossibleCows(int arr[],int n,int dist,int k)
    {
        int noCows=1,lastCow=arr[0];
        for(int i=1;i<n;i++)
        {
            if(arr[i]-lastCow>=dist)
            {
                noCows=noCows+1;
                lastCow=arr[i];
            }
        }
        return noCows>=k;
    }

Time Complexity :- O(nlogn) + O((max-min+1)*n)
Space Complexity :- O(1)

Optimal Approach :-
===================

int aggressiveCows(int[] arr, int k) {
        Arrays.sort(arr);
        int n=arr.length;
        int diff=arr[n-1]-arr[0];
        int low=1,high=diff;
        int ans=diff;
        while(low<=high)
        {
            int mid=(low+high)/2;
            if(isPossibleCows(arr,n,mid,k))
            {
                ans=mid;
                low=mid+1;
            }
            else
            {
                high=mid-1;
            }
        }
        return ans;
    }
    public boolean isPossibleCows(int arr[],int n,int dist,int k)
    {
        int noCows=1,lastCow=arr[0];
        for(int i=1;i<n;i++)
        {
            if(arr[i]-lastCow>=dist)
            {
                noCows=noCows+1;
                lastCow=arr[i];
            }
        }
        return noCows>=k;
    }
}

Time Complexity :- O(nlogn) + O(log2(max-min+1)*n)
Space Complexity :- O(1)
