
Input :- n=4

Output :- 2

Brute Force :-
==============

func(n)
{
 int ans=-1;
 for(int i=1;i<=n;i++)
 {
  long val=i*i;
  if(val<=n)
  {
   ans=i;
  }
  else
  {
   break;
  }
 }
 return ans;
}

Time Complexity :- O(n)
Space Complexity :- O(1)

Optimal Approach :-
===================

func(n)
{
 long low=1,high=n;
 long ans=-1;
 while(low<=high)
 {
  long mid=low+(high-low)/2;
  long val=mid*mid;
  if(val<=n)
  {
   ans=mid;
   low=mid+1;
  }
  else
  {
   high=mid-1;
  }
 }
 return ans;
}

Time Complexity :- O(log2n)
Space Complexity :- O(1)
