Output :- arr = [1,2,4,3]

func(arr,n)
{
 int pivotIndex=-1;
 for(int i=n-2;i>=0;i--)
 {
  if(arr[i]<arr[i+1])
  {
   pivotIndex=i;
   break;
  }
 }
if(pivotIndex==-1)
{
 reverse(arr,0,n-1);
 return;
}
for(int i=n-1;i>=0;i--)
{
 if(arr[pivotIndex]<arr[i])
 {
  swap(arr[pivotIndex],arr[i]);
  break;
 }
}
reverse(arr,pivotIndex+1,n-1);
}

Time Complexity :- O(3*n)
Space Complexity :- O(1)
