
Input :- arr = [[1,2,3],[4,5,6],[7,8,9]]

Output :- arr = [[7,4,1],[8,5,2],[9,6,3]]

Brute Force :-
==============

func(matrix)
{
  int n=matrix.length;
  int ans[][]=new int[n][n];
  for(int i=0;i<n;i++)
  {
    for(int j=0;j<n;j++)
    {
      ans[j][n-1-i]=matrix[i][j];
    }
  }
  return ans;
}

Time Complexity :- O(n^2)
Space Complexity :- O(n^2)

Optimal Approach :-
===================

func(matrix)
{
  int n=matrix.length;
  for(int i=0;i<n-1;i++)
  {
    for(int j=i+1;j<n;j++)
    {
      int temp=matrix[i][j];
      matrix[i][j]=matrix[j][i];
      matrix[j][i]=temp;
    }
  }
  for(int i=0;i<n;i++)
  {
    reverse(matrix[i]);
  }
}

reverse(int arr[])
{
  int start=0,end=arr.length-1;
  while(start<=end)
  {
    int temp=arr[start];
    arr[start]=arr[end];
    arr[end]=temp;
    start++;
    end--;
  }
}

Time Complexity :- O(N/2*N/2) + O(N*N/2)
Space Complexity :- O(1)
