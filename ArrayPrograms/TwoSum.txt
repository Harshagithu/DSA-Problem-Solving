
Input :- arr = [2,6,5,8,11] target = 13
Output :- Indexes = {0,4} { 2 + 11 = 13 or 5 + 8 = 13}

Brute Force :-
==============

func(arr,n,target)
{
 for(int i=0;i<n;i++)
 {
  for(int j=i+1;j<n;j++)
  {
    if(arr[i]+arr[j]==target)
    {
     return new int[]{i,j};
    }
 }
}
return new int[]{-1,-1};
}

Time Complexity :- O(n^2)
Space Complexity :- O(1)

Better Approach :-
==================

func(arr,n,target)
{
 HashMap<Integer,Integer> hm=new HashMap<>();
 for(int i=0;i<n;i++)
 {
   int rem=target-arr[i];
   if(hm.containsKey(rem))
   {
     return new int[]{hm.get(rem),i};
   }
   hm.put(arr[i],i);
}
return new int[]{-1,-1};
}

Time Complexity :- O(n*logn)
Space Complexity :- O(n)

Optimal Approach :-
===================

func(arr,n,target)
{
 Arrays.sort(arr);
 int i=0,j=n-1;
 while(i<=j)
 {
   int sum=arr[i]+arr[j];
   if(sum==target) return "YES";
   else if(sum<target) i++;
   else j++;
 }
return "NO";
}

Time Complexity :- O(n) + O(nlogn)
Space Complexity :- O(1)
