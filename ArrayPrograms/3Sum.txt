Output :- [[-1,-1,2],[-1,0,1]]

Brute Force :-
==============

List<List<Integer>> threeSum(int[] nums) {
        int n=nums.length;
        List<List<Integer>> res=new ArrayList<>();
        Set<List<Integer>> set=new HashSet<>();
        for(int i=0;i<n;i++)
        {
            for(int j=i+1;j<n;j++)
            {
                for(int k=j+1;k<n;k++)
                {
                    if(nums[i]+nums[j]+nums[k]==0)
                    {
                        List<Integer> arr=new ArrayList<>(Arrays.asList(nums[i],nums[j],nums[k]));
                        Collections.sort(arr);
                        if(!set.contains(arr))
                        {
                            res.add(arr);
                            set.add(arr);
                        }
                    }
                }
            }
        }
        return res;
    }

Time Complexity :- O(n^3) 
Space Complexity :- O(m) where, m is number of triplets.

List<List<Integer>> threeSum(int[] nums) {
        int n=nums.length;
        List<List<Integer>> res=new ArrayList<>();
        Set<List<Integer>> set=new HashSet<>();
        for(int i=0;i<n;i++)
        {
            Set<Integer> hs=new HashSet<>();
            for(int j=i+1;j<n;j++)
            {
                int target=-(nums[i]+nums[j]);
                if(hs.contains(target))
                {
                    List<Integer> arr=new ArrayList<>(Arrays.asList(nums[i],nums[j],target));
                    Collections.sort(arr);
                    if(!set.contains(arr))
                    {
                        res.add(arr);
                        set.add(arr);
                    }
                }
                hs.add(nums[j]);
            }
        }
        return res;
    }

Time Complexity :- O(n^2)
Space Complexity :- O(n) + O(m) where, m is number of triplets.

List<List<Integer>> threeSum(int[] nums) {
        int n=nums.length;
        List<List<Integer>> res=new ArrayList<>();
        Arrays.sort(nums);
        for(int i=0;i<n;i++)
        {
            if(nums[i]>0) break;
            if(i>0&&nums[i]==nums[i-1]) continue;
            int j=i+1;
            int k=n-1;
            while(j<k)
            {
                long sum=nums[i]+nums[j];
                sum+=nums[k];
                if(sum==0)
                {
                    List<Integer> arr=Arrays.asList(nums[i],nums[j],nums[k]);
                    res.add(arr);
                    while(j<k&&nums[j]==nums[j+1])
                    {
                        j++;
                    }
                    while(j<k&&nums[k]==nums[k-1])
                    {
                        k--;
                    }
                    j++;
                    k--;
                }
                else if(sum<0)
                {
                    j++;
                }
                else
                {
                    k--;
                }
            }
        }
        return res;
    }

Time Complexity :- O(n^2)
Space Complexity :- O(m) where, m is number of triplets.
