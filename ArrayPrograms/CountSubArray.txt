
Input :- arr = [1,2,3,-3,1,1,1,4,2,-3]

Output :- 8

Brute Force :-
==============

func(arr,n,k)
{
 int count=0;
 for(int i=0;i<n;i++)
 {
  for(int j=i;j<n;j++)
  {
   int sum=0;
   for(int k=i;k<=j;k++)
   {
    sum+=arr[k];
   }
   if(sum==k)
   {
     count++;
   }
  }
 }
 return count;
}

Time Complexity :- O(n^3)
Space Complexity :- O(1)

Better Approach :-
==================

func(arr,n,k)
{
 int count=0;
 for(int i=0;i<n;i++)
 {
  int sum=0;
  for(int j=i;j<n;j++)
  {
    sum+=arr[j];
    if(sum==k)
    {
      count++;
    }
  }
 }
 return count;
}

Time Complexity :- O(n^2)
Space Complexity :- O(1)

Optimal Approach :-
===================

func(arr,n,k)
{
 Map<Integer,Integer> hm=new HashMap<>();
 int count=0;
 int prefixSum=0;
 hm.put(0,1);
 for(int i=0;i<n;i++)
 {
  prefixSum+=arr[i];
  if(hm.containsKey(prefixSum-k))
  {
    count++;
  }
  hm.put(prefixSum,hm.getOrDefault(prefixSum,0)+1);
 }
 return count;
}

Time Complexity :- O(n*logn)
Space Complexity :- O(n)
