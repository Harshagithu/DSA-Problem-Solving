
Input :- [1,2,6,8,7,9,3,4,5]
Output :- 8

Brute Force :-
==============

func(arr,n)
{
  Arrays.sort(arr);
  int largest=arr[n-1];
  for(int i=n-2;i>=0;i--)
  {
    if(arr[i]!=largest)
    {
      return arr[i];
    }
  }
 return -1;
}

Time Complexity :- O(nlogn) + O(n)
Space Complexity :- O(1)

func(arr,n)

  {
    if(arr[i]>largest)
    {
      largest=arr[i];

 for(int i=0;i<n;i++)
 {

 }
 return slargest;
}

func(arr,n)
{
  int largest=arr[0],slargest=-1;
  for(int i=1;i<n;i++)
  {
    if(arr[i]>largest)
    {
     slargest=largest;
     largest=arr[i];
    }
    else if(arr[i]<largest&&slargest<arr[i])
    {
      slargest=arr[i];
    }
  }
  return slargest;
}
