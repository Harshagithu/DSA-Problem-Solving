func(arr,n)
{
  int count=0;
  for(int i=0;i<n;i++)
  {
    if(arr[i]==0)
    {
     count=0;
    }
    else
    {
     count++;
    }
   maxones=Math.max(maxones,count);
 }
 return maxones;
}
