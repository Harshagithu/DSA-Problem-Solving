      }

    }

void func(int index,int src[],int sum,int k,int n, ArrayList<Integer> arr)
{
  if(index==n)
    {
      if(sum==k)
        {
          for(int i:arr) sop(i+" ");
        }
      return;
    }
  
  arr.add(src[index]);
  sum+=src[index];
  
  func(index+1,src,sum,k,n,arr);
  
  arr.remove(index);
  sum-=src[index];
  
  func(index+1,src,sum,k,n,arr);
}

bool func(int index,int src[],int sum,int k,int n, ArrayList<Integer> arr)
{
  if(index==n)
    {
      if(sum==k)
        {
          for(int i:arr) sop(i+" ");
          return true;
        }
      return false;
    }
  
  arr.add(src[index]);
  sum+=src[index];
  
  if(func(index+1,src,sum,k,n,arr)==true)return true;
  
  arr.remove(index);
  sum-=src[index];
  
  if(func(index+1,src,sum,k,n,arr))return true;
  
  return false;
}

int func(int index,int src[],int sum,int k,int n)
{
  if(index==n)
    {
      if(sum==k)
        {
          return 1;
        }
      return 0;
    }
  
  sum+=src[index];
  
  int left=func(index+1,src,sum,k,n);
  
  sum-=src[index];
  
  int right=func(index+1,src,sum,k,n);

  return left + right;
}

Time Complexity :- O(2^n)
Space Complexity :- O(n)
