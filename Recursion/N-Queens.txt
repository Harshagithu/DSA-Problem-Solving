class Solution {
    public List<List<String>> solveNQueens(int n) {
        List<List<String>> res=new ArrayList<>();
        char board[][]=new char[n][n];
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<n;j++)
            {
                board[i][j]='.';
            }
        }
        solve(0,n,board,new ArrayList<>(),res);
        return res;
    }
    public void solve(int index,int n,char board[][],ArrayList<String> arr,List<List<String>> res)
    {
        if(index==n)
        {
            for(int i=0;i<n;i++)
            {
                String str="";
                for(int j=0;j<n;j++)
                {
                    str+=board[i][j];
                }
                System.out.println(str);
                arr.add(str);
            }
            res.add(arr);
            arr.clear();
            return;
        }
        for(int j=0;j<n;j++)
        {
            if(isValid(board,index,j,n))
            {
                board[index][j]='Q';
                solve(index+1,n,board,arr,res);
            }
            board[index][j]='.';
        }
    }
    public boolean isValid(char board[][],int i,int j,int n)
    {
        for(int row=i;row>0;row--)
        {
            if(board[row][j]=='Q')return false;
        }
        for(int row=i-1,col=j-1;row>0&&col>0;row--,col--)
        {
            if(board[row][col]=='Q')return false;
        }
        for(int row=i-1,col=j+1;row>0&&col<n-1;row--,col++)
        {
            if(board[row][col]=='Q')return false;
        }
        return true;
    }

Time Complexity :- O(N!*N)
Space Complexity :- O(N^2)


}
